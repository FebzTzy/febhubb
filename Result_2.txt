    local DodgeVeil = { Attached = {}, lastTime = 0, Cooldown = 0.1, Distance = 6 }
    DodgeVeil.Trigger = function()
        if tick() - DodgeVeil.lastTime < DodgeVeil.Cooldown then return end
        DodgeVeil.lastTime = tick()
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local side = (math.random() > 0.5) and 1 or -1
        local offset = hrp.CFrame.RightVector * side * DodgeVeil.Distance
        hrp.CFrame = CFrame.new(hrp.Position + offset) * (hrp.CFrame - hrp.CFrame.Position)
    end
    DodgeVeil.Attach = function(kChar)
        if not kChar or DodgeVeil.Attached[kChar] then return end
        DodgeVeil.Attached[kChar] = true
        local humanoid = kChar:WaitForChild("Humanoid", 5)
        if not humanoid then return end
        local animator = humanoid:FindFirstChildOfClass("Animator") or humanoid:WaitForChild("Animator", 5)
        if not animator then return end
        local childAddedConn
        childAddedConn = humanoid.ChildAdded:Connect(function(child)
            if child:IsA("Animator") then
                DodgeVeil.Attached[kChar] = nil
                if childAddedConn then childAddedConn:Disconnect() end
                DodgeVeil.Attach(kChar)
            end
        end)
        kChar.AncestryChanged:Connect(function(_, parent)
            if not parent then
                DodgeVeil.Attached[kChar] = nil
                if childAddedConn then childAddedConn:Disconnect() end
            end
        end)
        animator.AnimationPlayed:Connect(function(track)
            if not Config.Surv_CrouchV then return end
            local animId = track.Animation and track.Animation.AnimationId or ""
            local id = animId:match("%d+")
            if id ~= "86266790353635" and id ~= "93136435416899" then return end
            local myChar = LocalPlayer.Character
            local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
            local kHRP = kChar:FindFirstChild("HumanoidRootPart")
            if not (myHRP and kHRP) then return end
            if (myHRP.Position - kHRP.Position).Magnitude > 100 then return end
            local flatDelta = Vector3.new(myHRP.Position.X - kHRP.Position.X, 0, myHRP.Position.Z - kHRP.Position.Z)
            if flatDelta.Magnitude <= 0 then return end
            local kLookFlat = Vector3.new(kHRP.CFrame.LookVector.X, 0, kHRP.CFrame.LookVector.Z)
            if kLookFlat.Magnitude <= 0.001 then return end
            if kLookFlat.Unit:Dot(flatDelta.Unit) >= 0.7 then DodgeVeil.Trigger() end
        end)
    end
    DodgeVeil.TryAttach = function(p)
        if p ~= LocalPlayer and p.Team and p.Team.Name == "Killer" and p.Character then
            DodgeVeil.Attach(p.Character)
        end
    end
    DodgeVeil.Setup = function(p)
        if p == LocalPlayer then return end
        p.CharacterAdded:Connect(function() DodgeVeil.TryAttach(p) end)

    end

    function enableSpoofer()
        if not Config.Misc_FakeName then return end
        local pg = LocalPlayer:FindFirstChild("PlayerGui")
        if not pg then return end
        local function applySpooferToObj(obj)
            if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
                local function updateText()
                    if not Config.Misc_FakeName then return end
                    local cur = obj.Text
                    local changed = false
                    if not obj:GetAttribute("OriginalText") then obj:SetAttribute("OriginalText", cur) end
                    for _, plr in ipairs(_GetPlayers()) do
                        if string.find(cur, plr.Name) or string.find(cur, plr.DisplayName) then
                            cur = string.gsub(cur, plr.Name, "Leenz")
                            cur = string.gsub(cur, plr.DisplayName, "Leenz")
                            changed = true
                        end
                    end
                    if changed and obj.Text ~= cur then obj.Text = cur end
                end
                updateText()
                table.insert(State.SpooferConns, obj:GetPropertyChangedSignal("Text"):Connect(updateText))
            elseif obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
                local function updateImage()
                    if not Config.Misc_FakeName then return end
                    local cur = obj.Image
                    local changed = false
                    for _, plr in ipairs(_GetPlayers()) do
                        if string.find(cur, tostring(plr.UserId)) then
                            if not obj:GetAttribute("OriginalImage") then obj:SetAttribute("OriginalImage", cur) end
                            changed = true
                            break
                        end
                    end
                    if changed and obj.Image ~= "rbxassetid://138338413066899" then
                        obj.Image = "rbxassetid://138338413066899"
                    end
                end
                updateImage()
                table.insert(State.SpooferConns, obj:GetPropertyChangedSignal("Image"):Connect(updateImage))
            end
        end
        for _, obj in ipairs(pg:GetDescendants()) do applySpooferToObj(obj) end
        table.insert(State.SpooferConns, pg.DescendantAdded:Connect(function(obj)
            task.defer(function() applySpooferToObj(obj) end)
        end))
    end

                            end
                            local cnt = 0

                                    or 0
                            end

                                lbl.Name = "MWD_CustomHookCounter"

                                lbl.TextStrokeTransparency = 0
                                lbl.TextScaled = true
                                lbl.Font = Enum.Font.SourceSansBold
                            end
                            lbl.Visible = true
                            if cnt >= 3 then
                                lbl.Text = "DEAD"
                                lbl.TextColor3 = Color3.fromRGB(255, 75, 75)
                            else
                                lbl.Text = "Hooks: " .. tostring(cnt)

        if tick() - _closestSurvivorCacheTime < 0.15 and _closestSurvivorCache ~= nil then
            if _closestSurvivorCache.Parent then return _closestSurvivorCache end
        end
        local myChar = player.Character

        local closestTarget = nil
        local cam = workspace.CurrentCamera
        local centerScreen = Vector2.new(cam.ViewportSize.X/2, cam.ViewportSize.Y/2)
        for _, p in ipairs(_GetPlayers()) do
            if p ~= player and p.Team and p.Team.Name == "Survivors" and p.Character then

                        end
                    end
                end
            end
        end

        _closestSurvivorCacheTime = tick()

                local m = v.Parent
                if m and not table.find(Cache.Pallets, m) then table.insert(Cache.Pallets, m) end

    function ClearESP(tipe)
        if tipe == "Player" then
            for _, p in pairs(_GetPlayers()) do
                if p.Character then
                    if p.Character:FindFirstChild("PEH") then p.Character.PEH:Destroy() end
                    if p.Character:FindFirstChild("PE_Text") then p.Character.PE_Text:Destroy() end
                end
            end
        elseif tipe == "Killer" then
            for _, p in pairs(_GetPlayers()) do
                if p.Character then
                    if p.Character:FindFirstChild("KEH") then p.Character.KEH:Destroy() end
                    if p.Character:FindFirstChild("KE_Text") then p.Character.KE_Text:Destroy() end
                end
            end
        elseif tipe == "Generator" then
            for _, v in pairs(Cache.Generators) do
                if v.model and v.model.Parent then
                    if v.model:FindFirstChild("GEH") then v.model.GEH:Destroy() end
                    if v.model:FindFirstChild("GE_Text") then v.model.GE_Text:Destroy() end
                end
            end
        elseif tipe == "Pallet" then
            for _, v in pairs(Cache.Pallets) do
                if v and v.Parent and v:FindFirstChild("PalletEH") then v.PalletEH:Destroy() end
            end
        elseif tipe == "Hook" then
            for _, v in pairs(Cache.Hooks) do
                if v and v.Parent and v:FindFirstChild("HookEH") then v.HookEH:Destroy() end
            end
        elseif tipe == "Window" then
            for _, v in pairs(Cache.Windows) do
                if v and v.Parent and v:FindFirstChild("WindowEH") then v.WindowEH:Destroy() end
            end
        elseif tipe == "Gate" then
            for _, v in pairs(Cache.Gates) do
                if v and v.Parent and v:FindFirstChild("GateEH") then v.GateEH:Destroy() end
            end
        elseif tipe == "SCP" then
            for _, v in pairs(Cache.SCPs) do
                if v and v.Parent and v:FindFirstChild("SCPEH") then v.SCPEH:Destroy() end
            end
        end
    end

    function UpdatePlayerESP()
        if not Config.ESP_Master then return end
        local myChar = LocalPlayer.Character
        local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
        if Config.ESP_KillerWarn and myHRP and GetRole() == "Survivor" then
            local warn = myHRP:FindFirstChild("KillerWarn")
            local closestDist = math.huge
            for _, p in ipairs(_GetPlayers()) do
                if p ~= LocalPlayer and IsKiller(p) and p.Character then
                    local kHrp = p.Character:FindFirstChild("HumanoidRootPart")
                    if kHrp then
                        local d = (kHrp.Position - myHRP.Position).Magnitude
                        if d < closestDist then closestDist = d end
                    end
                end
            end
            if closestDist <= 80 then
                if not warn then
                    warn = Instance.new("BillboardGui")
                    warn.Name = "KillerWarn"
                    warn.Size = UDim2.new(0, 30, 0, 30); warn.AlwaysOnTop = true
                    warn.StudsOffset = Vector3.new(0, 4, 0); warn.Parent = myHRP
                    local txt = Instance.new("TextLabel")
                    txt.Name = "WarnText"; txt.Size = UDim2.new(1,0,1,0)
                    txt.BackgroundTransparency = 1; txt.TextScaled = true
                    txt.TextStrokeTransparency = 0; txt.Font = Enum.Font.GothamBlack
                    txt.Parent = warn
                end
                local txt = warn:FindFirstChild("WarnText")
                if txt then
                    if closestDist <= 40 then txt.Text = "!!"; txt.TextColor3 = Color3.fromRGB(255,0,0)
                    else txt.Text = "!"; txt.TextColor3 = Color3.fromRGB(255,255,0) end
                end
            else
                if warn then warn:Destroy() end
            end
        else
            local warn = myHRP and myHRP:FindFirstChild("KillerWarn")
            if warn then warn:Destroy() end
        end
        if Config.ESP_Player or Config.ESP_Killer then
            for _, p in pairs(_GetPlayers()) do
                if p ~= LocalPlayer and p.Character then
                    local kStat = IsKiller(p)
                    local root = p.Character:FindFirstChild("Head")
                    if root and ((not kStat and Config.ESP_Player) or (kStat and Config.ESP_Killer)) then
                        local distance = myHRP and math.floor((root.Position - myHRP.Position).Magnitude) or 0
                        local col = kStat and Tuning.Colors.Killer or Tuning.Colors.Player
                        local hName = kStat and "KEH" or "PEH"
                        local tName = kStat and "KE_Text" or "PE_Text"
                        local hl = p.Character:FindFirstChild(hName)
                        if not hl then hl = Instance.new("Highlight"); hl.Name = hName; hl.Parent = p.Character end
                        hl.FillColor = col; hl.OutlineColor = col
                        hl.FillTransparency = Config.ESP_Outline and 1 or 0.5
                        hl.OutlineTransparency = 0
                        if Config.ESP_Name then
                            local eIcon = nil
                            if Config.ESP_ItemIcon and not kStat then
                                local eName = p:GetAttribute("EquippedItem")
                                if eName and eName ~= "" then eIcon = getItemIcon(eName) end
                            end
                            CreateModernESP(p.Character, tName, {
                                name = p.Name,
                                distance = Config.ESP_Distance and distance or nil,
                                color = col, icon = eIcon
                            })
                        else
                            local old = p.Character:FindFirstChild(tName)
                            if old then old:Destroy() end
                        end
                    end
                end
            end
        end
    end

    function UpdateSCPESP()
        if not Config.ESP_Master then return end
        if Config.ESP_SCP then
            for _, obj in ipairs(Cache.SCPs) do
                if obj and obj.Parent then
                    local hum = obj:FindFirstChildOfClass("Humanoid")

    -- ============ OPTIMIZED AIMBOT TARGET WITH CACHE ============
    local _aimbotTargetCache = nil
    local _aimbotCacheTime = 0
    function getBestAimbotTarget()
        if tick() - _aimbotCacheTime < 0.1 and _aimbotTargetCache ~= nil then
            if _aimbotTargetCache.Parent then return _aimbotTargetCache end
        end
        local best = nil
        local myRole = GetRole()
        local shortestFov = Config.Surv_Aimbot_Radius
        local shortest3D = math.huge
        local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
        for _, p in pairs(_GetPlayers()) do
            if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                local valid = false
                if myRole == "Survivor" and Config.Surv_Aimbot_Enabled and IsKiller(p) then valid = true
                elseif myRole == "Killer" and Config.Killer_Aimbot_Enabled and not IsKiller(p) then
                    local hum = p.Character:FindFirstChild("Humanoid")
                    if hum and hum.Health > 0 and not IsDowned(p.Character) then valid = true end
                end
                if valid then
                    local hrp = p.Character.HumanoidRootPart
                    local d3 = GetDistance(hrp.Position)
                    if myRole == "Survivor" and d3 <= Config.Surv_Aimbot_MaxDist then
                        local sp, on = Camera:WorldToViewportPoint(hrp.Position)
                        if on then
                            local d2 = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                            if d2 < shortestFov then shortestFov = d2; best = hrp end
                        end
                    elseif myRole == "Killer" and d3 <= Config.Killer_Aimbot_MaxDist then
                        if d3 < shortest3D then shortest3D = d3; best = hrp end
                    end
                end
            end
        end
        _aimbotTargetCache = best
        _aimbotCacheTime = tick()
        return best
    end

function IsKillerFacingMe(kChar, myPos, angleDeg)
    local kRoot = kChar and kChar:FindFirstChild("HumanoidRootPart")
    if not kRoot or not myPos then return false end

    local angle = math.rad(tonumber(angleDeg) or 60)
    local toMe = myPos - kRoot.Position
    if toMe.Magnitude < 0.05 then return true end

    toMe = Vector3.new(toMe.X, 0, toMe.Z)
    if toMe.Magnitude < 0.05 then return true end
    toMe = toMe.Unit

    local look = kRoot.CFrame.LookVector
    look = Vector3.new(look.X, 0, look.Z)
    if look.Magnitude <= 0.05 then return false end
    look = look.Unit

    local bodyDot = look:Dot(toMe)
    local head = kChar:FindFirstChild("Head")
    local headDot = bodyDot
    if head then
        local hLook = head.CFrame.LookVector
        hLook = Vector3.new(hLook.X, 0, hLook.Z)
        if hLook.Magnitude > 0.05 then
            headDot = hLook.Unit:Dot(toMe)
        end
    end

    return (bodyDot >= math.cos(angle)) or (headDot >= math.cos(angle))
end

    function gatherGates()
        local gates = {}

        for _, obj in pairs(map:GetDescendants()) do

    function setBypassGate(state)
        BypassGateEnabled = state
        local gates = gatherGates()
        for _, gate in pairs(gates) do
            local lg = gate:FindFirstChild("LeftGate"); local rg = gate:FindFirstChild("RightGate")
            local le = gate:FindFirstChild("LeftGate-end") or gate:FindFirstChild("LeftGate-end2")
            local re = gate:FindFirstChild("RightGate-end") or gate:FindFirstChild("RightGate-end2")
            local box = gate:FindFirstChild("Box")
            if state then
                if lg then lg.Transparency = 1; lg.CanCollide = false end
                if rg then rg.Transparency = 1; rg.CanCollide = false end
                if le then le.Transparency = 0 end
                if re then re.Transparency = 0 end
                if box then box.CanCollide = false end
            else
                if lg then lg.Transparency = 0; lg.CanCollide = true end
                if rg then rg.Transparency = 0; rg.CanCollide = true end
                if le then le.Transparency = 1 end
                if re then re.Transparency = 1 end
                if box then box.CanCollide = true end
            end
        end
    end

    -- ============ FLEE KILLER ============
    function GetNearestKiller()
        local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        if not root then return nil, math.huge end
        local nk, nd = nil, math.huge
        for _, plr in ipairs(_GetPlayers()) do
            if plr ~= player and plr.Team and plr.Team.Name == "Killer"
            and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                local d = (plr.Character.HumanoidRootPart.Position - root.Position).Magnitude
                if d < nd then nd = d; nk = plr end
            end
        end
        return nk, nd
    end
    function FleeKiller()
        if not FleeKillerEnabled then return end
        local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        if not root then return end
        local killer, distance = GetNearestKiller()
        if not killer or distance > FleeDistance then return end
        if tick() - FleeCooldown < 3 then return end
        local kr = killer.Character and killer.Character:FindFirstChild("HumanoidRootPart")
        if not kr then return end
        local dir = (root.Position - kr.Position).Unit
        local sp = root.Position + dir * 80
        root.CFrame = CFrame.new(sp)
        FleeCooldown = tick()
    end

    AutoPalletEnabled = false
    AutoPalletSafety = true
    AutoPalletDistance = 10
    local _lastPalletDrop = 0
    local _usedPallets = {}
    local _autoPalletConn = nil
    function GetKillerRoot()
        for _, plr in ipairs(_GetPlayers()) do
            if plr == LocalPlayer then continue end
            if plr.Team and plr.Team.Name == "Survivors" then continue end
            local c = plr.Character
            if c then local r = c:FindFirstChild("HumanoidRootPart"); if r then return r end end
        end
        return nil

    function IsSafeToDropPallet(char)
        if not AutoPalletSafety then return true end
        if not char then return false end
        if char:GetAttribute("Knocked") == true then return false end
        if char:GetAttribute("IsCarried") == true then return false end
        if char:GetAttribute("IsHooked") == true then return false end
        local i = char:FindFirstChild("CheckInterractable")
        if i then
            if i:GetAttribute("isVaulting") or i:GetAttribute("isRepairing")
            or i:GetAttribute("isUnhooking") or i:GetAttribute("isHealing")
            or i:GetAttribute("isSliding") or i:GetAttribute("isCarrying") then return false end
        end
        local h = char:FindFirstChildOfClass("Humanoid")
        if h and h.Health <= 0 then return false end
        return true
    end
    function FindNearestPallet(myRoot)
        local bp, bd = nil, AutoPalletDistance
        for _, pallet in ipairs(Cache.Pallets or {}) do
            if not pallet or not pallet.Parent then continue end
            if _usedPallets[pallet] then continue end
            local rp = pallet:FindFirstChild("PrimaryPartPallet") or pallet:FindFirstChild("PalletPoint")
                or pallet:FindFirstChild("PalletPointSlide") or pallet:FindFirstChildWhichIsA("BasePart")
            if not rp then continue end
            local ok, pos = pcall(function() return rp.Position end)
            if not ok or not pos then continue end
            local d = (myRoot.Position - pos).Magnitude
            if d < bd then bd = d; bp = pallet end
        end
        return bp
    end
    function DropPallet(pallet)
        if not pallet then return false end
        local t = pallet:FindFirstChild("PalletPointSlide") or pallet:FindFirstChild("PalletPoint") or pallet:FindFirstChildWhichIsA("BasePart")
        if not t then return false end
        local r = ReplicatedStorage:FindFirstChild("Remotes")
        local de = r and r:FindFirstChild("Pallet") and r.Pallet:FindFirstChild("PalletDropEvent")
        if de then
            pcall(function() de:FireServer(t) end)
            _usedPallets[pallet] = true; _lastPalletDrop = tick()
            return true
        end
        return false
    end
    function StartAutoPallet()
        if _autoPalletConn then return end
        _autoPalletConn = task.spawn(function()
            while AutoPalletEnabled do
                task.wait(0.4)
                if GetRole() ~= "Survivor" then continue end
                if tick() - _lastPalletDrop < 2.5 then continue end
                pcall(function()
                    local char = LocalPlayer.Character
                    local mr = char and char:FindFirstChild("HumanoidRootPart")
                    local h = char and char:FindFirstChildOfClass("Humanoid")
                    if not mr or not h or h.Health <= 0 then return end
                    if not IsSafeToDropPallet(char) then return end
                    local kr = GetKillerRoot(); if not kr then return end
                    if (mr.Position - kr.Position).Magnitude > AutoPalletDistance then return end
                    local p = FindNearestPallet(mr)
                    if p then DropPallet(p) end
                end)
            end
        end)
    end
    function StopAutoPallet()
        if _autoPalletConn then task.cancel(_autoPalletConn); _autoPalletConn = nil end
    end
    LocalPlayer.CharacterAdded:Connect(function()
        _usedPallets = {}; _lastPalletDrop = 0
    end)

        function isTargetVisible(o, t, tc)
            local d = t - o; local dist = d.Magnitude
            if dist < 0.1 then return true end
            local rp = RaycastParams.new()
            rp.FilterType = Enum.RaycastFilterType.Exclude
            local ex = {}
            local lc = LocalPlayer.Character
            if lc then table.insert(ex, lc) end
            if tc and tc ~= lc then table.insert(ex, tc) end
            if LaserBeam then table.insert(ex, LaserBeam) end
            rp.FilterDescendantsInstances = ex
            return workspace:Raycast(o, d.Unit * dist, rp) == nil
        end
        local SCPCache = {}; local SCPCacheTimer = 0
        function GetSCPs()
            if tick() - SCPCacheTimer < 0.5 then return SCPCache end
            local nt = {}
            local mf = workspace:FindFirstChild("Map")
            if mf then
                for _, c in pairs(mf:GetDescendants()) do
                    if c:IsA("Model") then
                        local attr = c:GetAttributes()
                        if c:GetAttribute("CorpseCreated0492") or next(attr) ~= nil then

        local function getTargetVelocity(tc)
            local rp = tc:FindFirstChild("HumanoidRootPart"); if not rp then return Vector3.zero end
            local cp = rp.Position
            local ca = velocityCache[tc]
            if ca then
                local td = tick() - ca.time
                if td > 0.001 and td < 0.1 then
                    local rv = (cp - ca.pos) / td
                    local sv = ca.vel:Lerp(rv, PREDICT_SETTINGS.LerpSmoothness)
                    velocityCache[tc] = { pos = cp, vel = sv, time = tick() }
                    return sv
                end
            end
            velocityCache[tc] = { pos = cp, vel = rp.Velocity, time = tick() }
            return rp.Velocity
        end
        local function GetPredictedToFPosition(op, tp, tc, bs)
            bs = bs or 500
            local tpos = tp.Position
            local d = (tpos - op).Magnitude
            if d < 8 then return tpos end
            local tv = getTargetVelocity(tc)
            local rp = tpos - op
            local a = tv:Dot(tv) - (bs*bs)
            local b = 2 * rp:Dot(tv)
            local c = rp:Dot(rp)
            local t = nil
            if math.abs(a) < 0.01 then
                if math.abs(b) > 0.01 then t = -c/b end
            else
                local disc = b*b - 4*a*c
                if disc >= 0 then
                    local sq = math.sqrt(disc)
                    local t1 = (-b - sq) / (2*a)
                    local t2 = (-b + sq) / (2*a)
                    if t1 > 0.001 then t = t1 elseif t2 > 0.001 then t = t2 end
                end
            end
            local fp = tpos
            if t then
                local nv = tv * PREDICT_SETTINGS.PredictionEfficiency
                fp = tpos + (nv * t)
                if PREDICT_SETTINGS.EnableJitter then
                    fp = fp + Vector3.new((math.random()-0.5)*PREDICT_SETTINGS.MaxJitterStuds, (math.random()-0.5)*(PREDICT_SETTINGS.MaxJitterStuds/2), (math.random()-0.5)*PREDICT_SETTINGS.MaxJitterStuds)
                end
            end
            return fp
        end

                if not silentAimPredict then return (tpos - co).Unit, g, co, tpos end
                local pp = GetPredictedToFPosition(co, tp, tc, 500)
                return (pp - co).Unit, g, co, pp
            end

    local AntiSlowVault = {
        Hooked = false,
        OrigFacing = nil,
        OrigVaultAnim = nil,
        Controller = nil,
    }

    function AntiSlowVault.Apply()
        if AntiSlowVault.Hooked then return true end

        local ok, result = pcall(function()
            local Modules = ReplicatedStorage:WaitForChild("Modules", 5)
            if not Modules then return nil end
            local Survivors = Modules:WaitForChild("Survivors", 5)
            if not Survivors then return nil end
            local Controller = Survivors:WaitForChild("SurvivorAnimationsController", 5)
            if not Controller then return nil end
            return require(Controller)
        end)

        if not ok or not result then return false end

        AntiSlowVault.Controller = result

        if result.__LEENZ_HOOKED then
            AntiSlowVault.Hooked = true
            return true
        end

        if result._isFacingStraightEnough and not AntiSlowVault.OrigFacing then
            AntiSlowVault.OrigFacing = result._isFacingStraightEnough
            result._isFacingStraightEnough = function(self, part1, part2, maxAngle)
                if Config.Surv_PerfectVault then
                    self.characterspeed = 20
                    return true, 0
                end
                return AntiSlowVault.OrigFacing(self, part1, part2, maxAngle)
            end
        end

        if result._onVaultAnimation and not AntiSlowVault.OrigVaultAnim then
            AntiSlowVault.OrigVaultAnim = result._onVaultAnimation
            result._onVaultAnimation = function(self, vaultPoint, isSprinting)
                if Config.Surv_PerfectVault then
                    return AntiSlowVault.OrigVaultAnim(self, vaultPoint, true)
                end
                return AntiSlowVault.OrigVaultAnim(self, vaultPoint, isSprinting)
            end
        end

        result.__LEENZ_HOOKED = true
        AntiSlowVault.Hooked = (AntiSlowVault.OrigFacing ~= nil) or (AntiSlowVault.OrigVaultAnim ~= nil)
        return AntiSlowVault.Hooked
    end

            tries = tries + 1
            if AntiSlowVault.Apply() then
                print("[Anti Slow Vault] Hooked on attempt " .. tries)
                break
            end
            task.wait(1)
        end
    end)

    -- ============ SPEAR INTERCEPTOR (Killer) ============
    function setupSpearInterceptor()
        if SpearInterceptorHooked then return end
        if not getrawmetatable or not setreadonly then return end
        local Spearthrow = nil
        pcall(function() Spearthrow = ReplicatedStorage.Remotes.Killers.Veil.Spearthrow end)
        local mt = getrawmetatable(game)
        setreadonly(mt, false)
        local oldNC = mt.__namecall
        mt.__namecall = newcclosure(function(self, ...)
            local m = getnamecallmethod()
            if m == "FireServer" and not checkcaller() and typeof(self) == "Instance"
            and self.ClassName == "RemoteEvent" and self.Name == "Spearthrow" then
                if AimConfig.Aim_SilentVeil and not AimConfig.Aim_SilentVeilV2 then return nil end
                if AimConfig.Aim_SilentVeilV2 and not isFiringSpear then
                    local lookVec, speed, originPos = ...
                    speed = speed or AimConfig.SPEAR_Speed or 165
                    local myC = LocalPlayer.Character
                    local mh = myC and myC:FindFirstChild("HumanoidRootPart")
                    local sp = myC and (myC:FindFirstChild("Head") or mh)
                    local isSp = myC and myC:GetAttribute("special") == true
                    if Config.SpearSmart_enable then speed = isSp and 165 or 142.5
                    else speed = AimConfig.SPEAR_Speed or 165 end
                    originPos = originPos or (Config.SpearSmart_enable and mh and mh.Position) or (sp and sp.Position)
                    local bd = lookVec
                    local tp = getClosestSurvivor()
                    if tp and originPos then
                        local th = tp:IsA("Model") and tp:FindFirstChild("HumanoidRootPart") or tp
                        local tpos = th.Position
                        local tv = Vector3.new(0,0,0)
                        local thum = tp.Parent and tp.Parent:FindFirstChildOfClass("Humanoid")
                        if thum and thum.MoveDirection.Magnitude > 0 then tv = thum.MoveDirection * thum.WalkSpeed
                        elseif th:IsA("BasePart") then tv = th.AssemblyLinearVelocity end
                        tv = Vector3.new(tv.X, 0, tv.Z)
                        local d = (tpos - originPos).Magnitude
                        local tth = d / math.max(speed, 1)
                        if Config.SpearSmart_enable then
                            local lm = AimConfig.Veil_LeadMultiplier or 1.4
                            local pp = tpos + (tv * (tth * lm))
                            local g = workspace.Gravity * 0.5
                            local drop = 0.5 * g * (tth * tth)
                            local fp = pp + Vector3.new(0, drop - 1.5, 0)
                            bd = (fp - originPos).Unit
                        else
                            local dp = math.clamp(d / 50, 0.1, 4.0)
                            local pp = tpos + (tv * (tth * dp))
                            local dm = math.clamp(d / 100, 1, 2.5)
                            local ag = math.max(0, d - 8)
                            local g = AimConfig.AIM_Auto and ag or (AimConfig.SPEAR_Gravity or workspace.Gravity * 0.5)
                            local drop = 0.5 * g * (tth * tth) * dm
                            local fp = pp + Vector3.new(0, drop, 0)
                            bd = (fp - originPos).Unit
                        end
                    end
                    isFiringSpear = true
                    pcall(function()
                        if Spearthrow then Spearthrow:FireServer(bd, speed, originPos)
                        else self:FireServer(bd, speed, originPos) end
                    end)
                    isFiringSpear = false
                    return
                end
            end
            return oldNC(self, ...)
        end)
        setreadonly(mt, true)
        SpearInterceptorHooked = true
    end

    -- ============ KILLER PERKS DISPLAY ============
    KillerPerksEnabled = false
    KillerPerksGui = nil
    KillerPerksMinimized = false

    function CreateKillerPerksGUI()
        if KillerPerksGui then pcall(function() KillerPerksGui:Destroy() end); KillerPerksGui = nil end
        if not KillerPerksEnabled then return end

        local function esc(t)
            t = tostring(t or "")
            t = t:gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;")
            return t
        end
        local function parsePerk(name)
            name = tostring(name or "")
            local pn, lvl = name:match("^(.+)%s+(%d+)$")
            if not pn then return nil end
            pn = pn:gsub("^%s+", ""):gsub("%s+$", "")
            if pn == "" then return nil end
            local ex = { head=true, torso=true, humanoid=true, humanoidrootpart=true, ["left arm"]=true, ["right arm"]=true, ["left leg"]=true, ["right leg"]=true }
            if ex[pn:lower()] then return nil end
            return pn, lvl
        end
        local function getKiller()
            for _, plr in ipairs(_GetPlayers()) do
                local tn = plr.Team and plr.Team.Name
                if tn and tn:lower():find("killer") then return plr end
            end
            return nil
        end
        local function readPerks(k)
            if not k then return {} end
            local c = k.Character or workspace:FindFirstChild(k.Name) or workspace:FindFirstChild(k.DisplayName)
            if not c then return {} end
            local r, seen = {}, {}
            for _, ch in ipairs(c:GetChildren()) do
                local pn, lvl = parsePerk(ch.Name)
                if pn and not seen[pn] then seen[pn] = true; table.insert(r, {Name=pn, Level=lvl}) end
            end
            table.sort(r, function(a,b) return tostring(a.Name) < tostring(b.Name) end)
            return r
        end
        local function buildText()
            local k = getKiller()
            local kn = k and (k.DisplayName or k.Name) or "Unknown"
            local perks = readPerks(k)
            if #perks == 0 then
                for _, plr in ipairs(_GetPlayers()) do
                    local cand = readPerks(plr)
                    if #cand > 0 then k = plr; kn = plr.DisplayName or plr.Name; perks = cand; break end

    function TeleportToPart(part)
        if not part then return end
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            local offset = Vector3.new(0, 3, 0)
            if part:IsA("BasePart") then
                hrp.CFrame = part.CFrame + offset
            elseif part:IsA("Model") then
                local p = part:FindFirstChildWhichIsA("BasePart")
                if p then hrp.CFrame = p.CFrame + offset end
            end
        end
    end

        if #Cache.Generators == 0 then ScanMap() end

    -- ============ SERVER TOOLS ============
    MinPlayers, MaxPlayers = 1, 6
    autoReconnect = true
    function ServerHop()
        local pid = game.PlaceId
        local servers, cursor = {}, ""
        repeat
            local url = "https://games.roblox.com/v1/games/" .. pid .. "/servers/Public?sortOrder=Asc&limit=100" .. (cursor ~= "" and "&cursor=" .. cursor or "")
            local s, r = pcall(function() return HttpService:JSONDecode(game:HttpGet(url)) end)
            if s and r and r.data then
                for _, sv in ipairs(r.data) do
                    if sv.id ~= game.JobId and sv.playing >= MinPlayers and sv.playing <= MaxPlayers and sv.playing < sv.maxPlayers then
                        table.insert(servers, sv)
                    end
                end
                cursor = r.nextPageCursor
            else break end
        until not cursor or #servers > 0
        if #servers > 0 then
            local t = servers[math.random(#servers)]
            task.wait(0.3)
            TeleportService:TeleportToPlaceInstance(pid, t.id, LocalPlayer)
        else
            warn("No Server found, try again")
        end
    end
    LocalPlayer.OnTeleport:Connect(function(state)
        if state == Enum.TeleportState.Failed and autoReconnect then
            task.wait(3)
            TeleportService:Teleport(game.PlaceId, LocalPlayer)
        end
    end)

        W.GB_GetPoints = function(m)
            local pts = {}
            if m then
                for _, o in ipairs(m:GetChildren()) do
                    if o:IsA("BasePart") and string.find(o.Name, "GeneratorPoint") then
                        table.insert(pts, o)
                    end
                end
            end
            return pts
        end
    end

    function W.SU_IsHooked()
        local char = LocalPlayer.Character
        if not char then return false end
        return char:GetAttribute("IsHooked") == true or char:GetAttribute("isHooked") == true
            or char:GetAttribute("Hooked") == true or char:GetAttribute("HookedState") == true
    end

    function W.SU_GetRandomGeneratorPoint()
        local gens = W.GB_GetAllGenerators()
        if not gens or #gens == 0 then return nil end
        local valid = {}
        for _, g in ipairs(gens) do
            if g and g.Parent then
                for _, p in ipairs(W.GB_GetPoints(g)) do
                    if p and p.Parent then table.insert(valid, p) end
                end
            end
        end
        if #valid == 0 then return nil end
        return valid[math.random(1, #valid)]
    end

    function W.SU_GetKillerHRP()
        for _, p in ipairs(_GetPlayers()) do
            if p ~= LocalPlayer and SU_TeamIs(p, "Killer") and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                if hrp then return hrp end
            end
        end
        return nil
    end

    function W.SU_Abort()
        SU.Following = false
        if SU._activeThread and coroutine.status(SU._activeThread) ~= "dead" then
            pcall(function() task.cancel(SU._activeThread) end)
        end
        SU._activeThread = nil
    end

    function W.SU_Trigger()
        if SU.Following then return end
        local now = tick()
        if now - SU._lastTrigger < SU._cooldown then return end
        if not W.SU_IsHooked() then return end
        SU._lastTrigger = now; SU.Following = true
        SU.TriggerCount = SU.TriggerCount + 1
        local char = LocalPlayer.Character
        if char then
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if hrp then SU._hookPos = hrp.Position; SU._hookCFrame = hrp.CFrame end
        end
        SU._activeThread = task.spawn(function()
            local c = LocalPlayer.Character
            if not c then SU.Following = false; SU._activeThread = nil; return end
            local hrp = c:FindFirstChild("HumanoidRootPart")
            if not hrp then SU.Following = false; SU._activeThread = nil; return end
            local re = ReplicatedStorage:FindFirstChild("Remotes")
                and ReplicatedStorage.Remotes:FindFirstChild("Generator")
                and ReplicatedStorage.Remotes.Generator:FindFirstChild("RepairEvent")
            if not W.SU_IsHooked() then SU.Following = false; SU._activeThread = nil; return end
            local tp = W.SU_GetRandomGeneratorPoint()
            if tp then
                pcall(function() hrp.Velocity = Vector3.zero; hrp.CFrame = tp.CFrame + Vector3.new(0, 3, 0) end)
                task.wait(0.15)
                if re then
                    pcall(function() re:FireServer(tp, true) end); task.wait(0.35)
                    pcall(function() re:FireServer(tp, false) end); task.wait(0.15)
                    pcall(function() re:FireServer(tp, true) end); task.wait(0.35)
                    pcall(function() re:FireServer(tp, false) end)
                end
            end
            local fUntil = tick() + (SU.FollowDuration or 30)
            local bOff = SU.FollowDistance or 20
            while tick() < fUntil and SU.Enabled do
                if not W.SU_IsHooked() then
                    SU.Following = false; SU._activeThread = nil
                    return
                end
                local cc = LocalPlayer.Character
                if not cc then break end
                local r = cc:FindFirstChild("HumanoidRootPart")
                if not r then break end
                local khrp = W.SU_GetKillerHRP()
                if khrp then
                    pcall(function()
                        r.Velocity = Vector3.zero
                        local tPos = Vector3.new(khrp.Position.X, khrp.Position.Y - bOff, khrp.Position.Z)
                        r.CFrame = CFrame.new(tPos, tPos + Vector3.new(0, 0, -1))
                    end)
                end
                task.wait(0.05)
            end
            if W.SU_IsHooked() then
                local c2 = LocalPlayer.Character
                if c2 then
                    local r2 = c2:FindFirstChild("HumanoidRootPart")
                    if r2 then
                        if SU._hookCFrame then pcall(function() r2.Velocity = Vector3.zero; r2.CFrame = SU._hookCFrame end)
                        elseif SU._hookPos then pcall(function() r2.Velocity = Vector3.zero; r2.CFrame = CFrame.new(SU._hookPos + Vector3.new(0, 3, 0)) end) end
                    end
                end
            end
            SU.Following = false; SU._activeThread = nil
        end)
    end

    function W.SU_Start()
        if SU.MonitorConn then return end
        SU._wasHooked = false
        SU.MonitorConn = RunService.Heartbeat:Connect(function()
            if not SU.Enabled then return end
            if _frameCounter % 5 ~= 0 then return end
            local char = LocalPlayer.Character
            if not char then return end
            local isH = W.SU_IsHooked()
            if not isH then SU._wasHooked = false; return end
            if not SU._wasHooked and not SU.Following then SU._wasHooked = true; W.SU_Trigger() end
        end)
    end

    function W.SU_Stop()
        if SU.MonitorConn then SU.MonitorConn:Disconnect(); SU.MonitorConn = nil end
        W.SU_Abort(); SU._wasHooked = false
    end

-- ============ CAMERA FOV HOLD (ANTI JITTER) ============
local FOVConnName = "Leenz_FOV_Hold"
local function StartFOVHold()
    pcall(function() RunService:UnbindFromRenderStep(FOVConnName) end)
    RunService:BindToRenderStep(FOVConnName, Enum.RenderPriority.Camera.Value + 1, function()
        if not FOVEnabled then return end
        local c = workspace.CurrentCamera
        if c and c.FieldOfView ~= TargetFOV then
            c.FieldOfView = TargetFOV
        end
    end)
end
local function StopFOVHold()
    pcall(function() RunService:UnbindFromRenderStep(FOVConnName) end)
end
